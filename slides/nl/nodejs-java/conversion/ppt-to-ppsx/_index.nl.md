Om het bestand PPT naar PPSX te converteren, moet u een presentatie maken van het bestand PPT en dit opslaan als PPSX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-code voor het converteren van PPT naar PPSX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.ppsx", aspose.slides.SaveFormat.Ppsx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
