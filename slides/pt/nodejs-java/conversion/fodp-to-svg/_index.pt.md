Para converter FODP para SVG, você precisará criar uma apresentação a partir do arquivo FODP e salvá-la como SVG.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter FODP em SVG" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.fodp");
try
{
    var slide = pres.getSlides().get_Item(0);
    var svgStream = java.newInstanceSync("java.io.FileOutputStream", "image.svg");
    slide.writeAsSvg(svgStream);
    svgStream.close();
}
finally
{
    if (pres != null) pres.dispose();
}
```
