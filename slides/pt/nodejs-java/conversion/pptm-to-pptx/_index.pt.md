Para converter PPTM para PPTX, você precisará criar uma apresentação a partir do arquivo PPTM e salvá-la como PPTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter PPTM em PPTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptm");
try
{
    pres.save("output.pptx", aspose.slides.SaveFormat.Pptx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
