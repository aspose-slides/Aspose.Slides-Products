Para converter POT para PPT, você precisará criar uma apresentação a partir do arquivo POT e salvá-la como PPT.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter POT em PPT" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pot");
try
{
    pres.save("output.ppt", aspose.slides.SaveFormat.Ppt);
}
finally
{
    if (pres != null) pres.dispose();
}
```
