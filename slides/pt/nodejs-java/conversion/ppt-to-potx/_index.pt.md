Para converter PPT para POTX, você precisará criar uma apresentação a partir do arquivo PPT e salvá-la como POTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter PPT em POTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.potx", aspose.slides.SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
