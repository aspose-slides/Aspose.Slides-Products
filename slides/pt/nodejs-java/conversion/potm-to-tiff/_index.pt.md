Para converter POTM para TIFF, você precisará criar uma apresentação a partir do arquivo POTM e salvá-la como TIFF.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter POTM em TIFF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.tiff", aspose.slides.SaveFormat.Tiff);
}
finally
{
    if (pres != null) pres.dispose();
}
```
