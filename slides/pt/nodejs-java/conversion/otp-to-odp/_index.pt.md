Para converter OTP para ODP, você precisará criar uma apresentação a partir do arquivo OTP e salvá-la como ODP.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter OTP em ODP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.otp");
try
{
    pres.save("output.odp", aspose.slides.SaveFormat.Odp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
