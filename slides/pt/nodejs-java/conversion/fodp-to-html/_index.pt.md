Para converter FODP para HTML, você precisará criar uma apresentação a partir do arquivo FODP e salvá-la como HTML.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter FODP em HTML" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.fodp");
try
{
    pres.save("output.html", aspose.slides.SaveFormat.Html5);
}
finally
{
    if (pres != null) pres.dispose();
}
```
