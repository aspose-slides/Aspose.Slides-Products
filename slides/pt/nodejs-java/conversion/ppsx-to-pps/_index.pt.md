Para converter PPSX para PPS, você precisará criar uma apresentação a partir do arquivo PPSX e salvá-la como PPS.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para converter PPSX em PPS" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppsx");
try
{
    pres.save("output.pps", aspose.slides.SaveFormat.Pps);
}
finally
{
    if (pres != null) pres.dispose();
}
```
