Per convertire PDF in PPS, dovrai creare la presentazione dal file PDF e salvarla come PPS.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Codice Node.js per convertire PDF in PPS" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.pps", aspose.slides.SaveFormat.Pps);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
