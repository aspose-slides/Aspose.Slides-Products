Untuk mengonversi OTP ke PPTM, Anda perlu membuat Presentasi dari file OTP dan menyimpannya sebagai PPTM.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Kode Node.js untuk mengubah OTP menjadi PPTM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.otp");
try
{
    pres.save("output.pptm", aspose.slides.SaveFormat.Pptm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
