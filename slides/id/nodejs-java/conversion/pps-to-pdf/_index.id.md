Untuk mengonversi PPS ke PDF, Anda perlu membuat Presentasi dari file PPS dan menyimpannya sebagai PDF.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Kode Node.js untuk mengubah PPS menjadi PDF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pps");
try
{
    pres.save("output.pdf", aspose.slides.SaveFormat.Pdf);
}
finally
{
    if (pres != null) pres.dispose();
}
```
