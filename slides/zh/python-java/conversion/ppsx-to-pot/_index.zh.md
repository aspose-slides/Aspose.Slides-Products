{{< blocks/products/pf/slides-design-update-10-2026/code title="将 PPSX 转换为 POT 的 Python 教程" >}}
```python

import jpype
import asposeslides

jpype.startJVM()

from asposeslides.api import Presentation, SaveFormat
from javax.imageio import ImageIO
from java.io import File

pres = Presentation("PowerPoint.ppsx");

pres.save("output.pot", SaveFormat.Pot);

jpype.shutdownJVM()
```
