Para convertir OTP a PPTX, deberá crear una presentación a partir del archivo OTP y guardarla como PPTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para convertir OTP en PPTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.otp");
try
{
    pres.save("output.pptx", aspose.slides.SaveFormat.Pptx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
