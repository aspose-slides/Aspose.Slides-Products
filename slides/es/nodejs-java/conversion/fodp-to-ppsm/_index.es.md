Para convertir FODP a PPSM, deberá crear una presentación a partir del archivo FODP y guardarla como PPSM.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para convertir FODP en PPSM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.fodp");
try
{
    pres.save("output.ppsm", aspose.slides.SaveFormat.Ppsm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
