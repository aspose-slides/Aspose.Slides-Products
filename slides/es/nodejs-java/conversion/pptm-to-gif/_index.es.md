Para convertir PPTM a GIF, deberá crear una presentación a partir del archivo PPTM y guardarla como GIF.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para convertir PPTM en GIF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptm");
try
{
    pres.save("output.gif", aspose.slides.SaveFormat.Gif);
}
finally
{
    if (pres != null) pres.dispose();
}
```
