Para convertir ODP a POTX, deberá crear una presentación a partir del archivo ODP y guardarla como POTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para convertir ODP en POTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.odp");
try
{
    pres.save("output.potx", aspose.slides.SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
