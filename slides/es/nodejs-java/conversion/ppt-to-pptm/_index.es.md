Para convertir PPT a PPTM, deberá crear una presentación a partir del archivo PPT y guardarla como PPTM.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para convertir PPT en PPTM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.pptm", aspose.slides.SaveFormat.Pptm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
