Para convertir POTX a PPSX, deberá crear una presentación a partir del archivo POTX y guardarla como PPSX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Código Node.js para convertir POTX en PPSX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potx");
try
{
    pres.save("output.ppsx", aspose.slides.SaveFormat.Ppsx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
