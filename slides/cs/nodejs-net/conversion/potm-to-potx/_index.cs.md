{{< blocks/products/pf/slides-design-update-10-2026/code title="JavaScript kód pro převod POTM do POTX" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.potx", SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
