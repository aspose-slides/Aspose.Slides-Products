{{< blocks/products/pf/slides-design-update-10-2026/code title="JavaScript kód pro převod PPSX do PPS" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.ppsx");
try
{
    pres.save("output.pps", SaveFormat.Pps);
}
finally
{
    if (pres != null) pres.dispose();
}
```
