{{< blocks/products/pf/slides-design-update-10-2026/code title="Kód Node.js pro převod PPSX do POTM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppsx");
try
{
    pres.save("output.potm", aspose.slides.SaveFormat.Potm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
