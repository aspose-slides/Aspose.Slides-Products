// Load first PPSX File
Presentation prest1 = new Presentation("prest1.ppsx");

// Load second PPSX File
Presentation prest2 = new Presentation("prest2.ppsx");

// Merge
for (ISlide slide : prest2.getSlides()) {
	// Merge from source to target
