{{< blocks/products/pf/slides-design-update-10-2026/code title="برنامج تعليمي لبايثون لتحويل OTP إلى TIFF" >}}
```python

import jpype
import asposeslides

jpype.startJVM()

from asposeslides.api import Presentation, SaveFormat
from javax.imageio import ImageIO
from java.io import File

pres = Presentation("PowerPoint.otp");

pres.save("output.tiff", SaveFormat.Tiff);

jpype.shutdownJVM()
```
