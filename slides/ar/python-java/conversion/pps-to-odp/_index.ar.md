{{< blocks/products/pf/slides-design-update-10-2026/code title="برنامج تعليمي لبايثون لتحويل PPS إلى ODP" >}}
```python

import jpype
import asposeslides

jpype.startJVM()

from asposeslides.api import Presentation, SaveFormat
from javax.imageio import ImageIO
from java.io import File

pres = Presentation("PowerPoint.pps");

pres.save("output.odp", SaveFormat.Odp);

jpype.shutdownJVM()
```
