لتحويل PDF إلى PPSM، ستحتاج إلى إنشاء عرض تقديمي من ملف PDF وحفظه باسم PPSM.

{{< blocks/products/pf/slides-design-update-10-2026/code title="كود Node.js لتحويل PDF إلى PPSM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.ppsm", aspose.slides.SaveFormat.Ppsm);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
