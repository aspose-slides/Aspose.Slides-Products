{{< blocks/products/pf/slides-design-update-10-2026/code title="كود جافا سكريبت لتحويل POTM إلى GIF" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.gif", SaveFormat.Gif);
}
finally
{
    if (pres != null) pres.dispose();
}
```
