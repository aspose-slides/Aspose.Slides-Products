{{< blocks/products/pf/slides-design-update-10-2026/code title="รหัส Node.js สำหรับการแปลง POTM เป็น PPSM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.ppsm", aspose.slides.SaveFormat.Ppsm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
