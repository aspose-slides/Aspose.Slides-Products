Чтобы преобразовать HTML в PDF, вам нужно будет создать презентацию из файла HTML и сохранить его как PDF.

{{< blocks/products/pf/slides-design-update-10-2026/code title="PHP-код для преобразования HTML в PDF" >}}
```php

<?php
require_once("http://localhost:8080/JavaBridge/java/Java.inc");
require_once("lib/aspose.slides.php");
        
$pres = new Presentation();
try
{
    $pres->getSlides()->removeAt(0);
    
    $filename = 'file.html';
    $f = fopen($filename, 'r');
    if ($f) {
        $contents = fread($f, filesize($filename));
        fclose($f);
    }
    
    $pres->getSlides()->addFromHtml($contents);        
    $pres->save("output.pdf", SaveFormat::Pdf);        
}
finally
{
    if ($pres != null) $pres->dispose();
}
?>
```
