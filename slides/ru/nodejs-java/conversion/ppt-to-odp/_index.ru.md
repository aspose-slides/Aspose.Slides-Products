Чтобы преобразовать PPT в ODP, вам нужно будет создать презентацию из файла PPT и сохранить ее как ODP.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Код Node.js для преобразования PPT в ODP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.odp", aspose.slides.SaveFormat.Odp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
