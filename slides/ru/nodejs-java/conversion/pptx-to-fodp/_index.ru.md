Чтобы преобразовать PPTX в FODP, вам нужно будет создать презентацию из файла PPTX и сохранить ее как FODP.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Код Node.js для преобразования PPTX в FODP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptx");
try
{
    pres.save("output.fodp", aspose.slides.SaveFormat.Fodp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
