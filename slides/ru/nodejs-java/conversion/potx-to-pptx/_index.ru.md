Чтобы преобразовать POTX в PPTX, вам нужно будет создать презентацию из файла POTX и сохранить ее как PPTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Код Node.js для преобразования POTX в PPTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potx");
try
{
    pres.save("output.pptx", aspose.slides.SaveFormat.Pptx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
