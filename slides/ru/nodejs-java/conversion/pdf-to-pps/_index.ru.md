Чтобы преобразовать PDF в PPS, вам нужно будет создать презентацию из файла PDF и сохранить ее как PPS.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Код Node.js для преобразования PDF в PPS" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.pps", aspose.slides.SaveFormat.Pps);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
