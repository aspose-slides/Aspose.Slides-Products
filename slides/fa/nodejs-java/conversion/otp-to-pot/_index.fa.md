{{< blocks/products/pf/slides-design-update-10-2026/code title="کد Node.js برای تبدیل OTP به POT" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.otp");
try
{
    pres.save("output.pot", aspose.slides.SaveFormat.Pot);
}
finally
{
    if (pres != null) pres.dispose();
}
```
