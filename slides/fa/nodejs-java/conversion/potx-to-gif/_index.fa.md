برای تبدیل POTX به GIF، باید Presentation را از فایل POTX ایجاد کنید و آن را به عنوان GIF ذخیره کنید.

{{< blocks/products/pf/slides-design-update-10-2026/code title="کد Node.js برای تبدیل POTX به GIF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potx");
try
{
    pres.save("output.gif", aspose.slides.SaveFormat.Gif);
}
finally
{
    if (pres != null) pres.dispose();
}
```
