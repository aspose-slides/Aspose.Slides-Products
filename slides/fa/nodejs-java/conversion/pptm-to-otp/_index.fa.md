برای تبدیل PPTM به OTP، باید Presentation را از فایل PPTM ایجاد کنید و آن را به عنوان OTP ذخیره کنید.

{{< blocks/products/pf/slides-design-update-10-2026/code title="کد Node.js برای تبدیل PPTM به OTP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptm");
try
{
    pres.save("output.otp", aspose.slides.SaveFormat.Otp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
