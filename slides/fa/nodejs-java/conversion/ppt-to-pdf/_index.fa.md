برای تبدیل PPT به PDF، باید Presentation را از فایل PPT ایجاد کنید و آن را به عنوان PDF ذخیره کنید.

{{< blocks/products/pf/slides-design-update-10-2026/code title="کد Node.js برای تبدیل PPT به PDF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.pdf", aspose.slides.SaveFormat.Pdf);
}
finally
{
    if (pres != null) pres.dispose();
}
```
