برای تبدیل POT به FODP، باید Presentation را از فایل POT ایجاد کنید و آن را به عنوان FODP ذخیره کنید.

{{< blocks/products/pf/slides-design-update-10-2026/code title="کد Node.js برای تبدیل POT به FODP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pot");
try
{
    pres.save("output.fodp", aspose.slides.SaveFormat.Fodp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
