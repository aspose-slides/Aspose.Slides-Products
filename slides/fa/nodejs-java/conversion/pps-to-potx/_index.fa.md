برای تبدیل PPS به POTX، باید Presentation را از فایل PPS ایجاد کنید و آن را به عنوان POTX ذخیره کنید.

{{< blocks/products/pf/slides-design-update-10-2026/code title="کد Node.js برای تبدیل PPS به POTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pps");
try
{
    pres.save("output.potx", aspose.slides.SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
