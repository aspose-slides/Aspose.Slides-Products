{{< blocks/products/pf/slides-design-update-10-2026/code title="C#-kod för att konvertera PDF till PPTX" >}}

```cs
{
    pres.Slides.AddFromPdf("InputPDF.pdf");
    pres.Save("OutputPresentation.pptx", SaveFormat.Pptx);
}
```
