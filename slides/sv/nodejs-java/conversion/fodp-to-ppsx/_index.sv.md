För att konvertera FODP till PPSX måste du skapa en presentation från filen FODP och spara den som PPSX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-kod för att konvertera FODP till PPSX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.fodp");
try
{
    pres.save("output.ppsx", aspose.slides.SaveFormat.Ppsx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
