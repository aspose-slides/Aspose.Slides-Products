För att konvertera POT till TIFF måste du skapa en presentation från filen POT och spara den som TIFF.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-kod för att konvertera POT till TIFF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pot");
try
{
    pres.save("output.tiff", aspose.slides.SaveFormat.Tiff);
}
finally
{
    if (pres != null) pres.dispose();
}
```
