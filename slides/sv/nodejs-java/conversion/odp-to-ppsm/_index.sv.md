För att konvertera ODP till PPSM måste du skapa en presentation från filen ODP och spara den som PPSM.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-kod för att konvertera ODP till PPSM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.odp");
try
{
    pres.save("output.ppsm", aspose.slides.SaveFormat.Ppsm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
