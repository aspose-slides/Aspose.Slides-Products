För att konvertera PDF till FODP måste du skapa en presentation från filen PDF och spara den som FODP.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-kod för att konvertera PDF till FODP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.fodp", aspose.slides.SaveFormat.Fodp);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
