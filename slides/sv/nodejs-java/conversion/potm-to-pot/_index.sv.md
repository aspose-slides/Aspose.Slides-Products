För att konvertera POTM till POT måste du skapa en presentation från filen POTM och spara den som POT.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-kod för att konvertera POTM till POT" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.pot", aspose.slides.SaveFormat.Pot);
}
finally
{
    if (pres != null) pres.dispose();
}
```
