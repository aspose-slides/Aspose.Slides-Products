För att konvertera POTM till PPT måste du skapa en presentation från filen POTM och spara den som PPT.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-kod för att konvertera POTM till PPT" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.ppt", aspose.slides.SaveFormat.Ppt);
}
finally
{
    if (pres != null) pres.dispose();
}
```
