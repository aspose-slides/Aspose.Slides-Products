{{< blocks/products/pf/slides-design-update-10-2026/code title="Cet exemple de code montre la conversion ODP en PNG C#" >}}
```cs
using (var presentation = new Presentation("template.odp"))
{
    // iterate over all slides in the presentation
    foreach (var slide in presentation.Slides)
    {
        // create a full scale image
        var bitmap = slide.GetThumbnail(1f, 1f);

        // save the image to disk in PNG format
        bitmap.Save(string.Format("Slide_{0}.png", slide.SlideNumber), System.Drawing.Imaging.ImageFormat.Png);
    }
} 

```
