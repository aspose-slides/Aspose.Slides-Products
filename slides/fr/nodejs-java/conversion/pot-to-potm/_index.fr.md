{{% blocks/products/pf/slides-design-update-10-2026/section h2="Convertir POT en POTM à l'aide de Node.js" variant="code" %}}

Pour convertir le POT en POTM, vous devrez créer une présentation à partir du fichier POT et l'enregistrer sous POTM.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Code Node.js pour convertir POT en POTM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pot");
try
{
    pres.save("output.potm", aspose.slides.SaveFormat.Potm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
