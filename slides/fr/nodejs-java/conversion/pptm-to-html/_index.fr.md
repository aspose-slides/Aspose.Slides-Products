Pour convertir le PPTM en HTML, vous devrez créer une présentation à partir du fichier PPTM et l'enregistrer sous HTML.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Code Node.js pour convertir PPTM en HTML" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptm");
try
{
    pres.save("output.html", aspose.slides.SaveFormat.Html5);
}
finally
{
    if (pres != null) pres.dispose();
}
```
