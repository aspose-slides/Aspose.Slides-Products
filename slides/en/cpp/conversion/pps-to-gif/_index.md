{{< blocks/products/pf/slides-design-update-10-2026/code title="PPS to GIF C++ Conversion Source Code" >}}
```cpp
auto presentation = MakeObject<Presentation>(u"presentation.pps");
auto slide = presentation->get_Slide(0);

auto desiredSize = Size(1200, 800);
auto image = slide->GetImage(desiredSize);
image->Save(u"slide.gif", ImageFormat::Gif);
image->Dispose();

presentation->Dispose();
```
