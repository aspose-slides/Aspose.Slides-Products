{{< blocks/products/pf/slides-design-update-10-2026/code title="PPSX to BMP C++ Conversion Source Code" >}}
```cpp
auto presentation = MakeObject<Presentation>(u"presentation.ppsx");
auto slide = presentation->get_Slide(0);

auto desiredSize = Size(1200, 800);
auto image = slide->GetImage(desiredSize);
image->Save(u"slide.bmp", ImageFormat::Bmp);
image->Dispose();

presentation->Dispose();
```
