```cs
using var presentation = new Presentation("template.pot");

foreach (var slide in presentation.Slides)
{
    using var fileStream = new FileStream($"Slide_{slide.SlideNumber}.svg", FileMode.Create);
    slide.WriteAsSvg(fileStream);
}
```
