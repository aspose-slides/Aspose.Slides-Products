{{< blocks/products/pf/slides-design-update-10-2026/code title="C# code to convert HTML to JPG" >}}
```cs
using var presentation = new Presentation();

var htmlContent = File.ReadAllText("page.html");
presentation.Slides.AddFromHtml(htmlContent);
presentation.Slides.RemoveAt(0);

for (var slideIndex = 0; slideIndex < presentation.Slides.Count; slideIndex++)
{
    var slide = presentation.Slides[slideIndex];
    using var image = slide.GetImage(1f, 1f);
    image.Save($"image-{slideIndex}.jpg", ImageFormat.Jpeg);
}
```
