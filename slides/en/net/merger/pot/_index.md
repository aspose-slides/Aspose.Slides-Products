{{% blocks/products/pf/agp/code-block title="Merge POT Files - C#" offSpacer="" %}}

```cs
using var firstPresentation = new Presentation("first.pot");
using var secondPresentation = new Presentation("second.pot");
using var mergedPresentation = new Presentation();

mergedPresentation.Slides.RemoveAt(0);

foreach (var slide in firstPresentation.Slides)
{
    mergedPresentation.Slides.AddClone(slide);
}

foreach (var slide in secondPresentation.Slides)
{
    mergedPresentation.Slides.AddClone(slide);
}

mergedPresentation.Save("merged.pot", SaveFormat.Pot);
```
