```php
$presentation = new Presentation();
try {
    $presentation->getSlides()->removeAt(0);
    $presentation->getSlides()->addFromPdf("document.pdf");
    $presentation->save("output.potm", SaveFormat::Potm);
} finally {
    $presentation->dispose();
}
```
