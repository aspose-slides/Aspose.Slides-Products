{{< blocks/products/pf/slides-design-update-10-2026/code title="PPTX to SVG Java Conversion Source Code" >}}
```java
Presentation presentation = new Presentation("presentation.pptx");
try {
    for (ISlide slide : presentation.getSlides()) {
        FileOutputStream svgOutputStream = new FileOutputStream("slide_" + slide.getSlideNumber() + ".svg");
        try {
            slide.writeAsSvg(svgOutputStream);
        } finally {
            svgOutputStream.close();
        }
    }
} finally {
    presentation.dispose();
}
```
