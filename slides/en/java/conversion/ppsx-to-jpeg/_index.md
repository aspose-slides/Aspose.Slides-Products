{{% blocks/products/pf/agp/code-block title="PPSX to JPEG Java Conversion Source Code" offSpacer="" %}}

```java
Presentation presentation = new Presentation("slideshow.ppsx");
try {
    for (ISlide slide : presentation.getSlides()) {
        IImage slideImage = slide.getImage(1f, 1f);
        try {
            slideImage.save("slide_" + slide.getSlideNumber() + ".jpeg", ImageFormat.Jpeg);
        } finally {
            slideImage.dispose();
        }
    }
} finally {
    presentation.dispose();
}
```
