{{< blocks/products/pf/slides-design-update-10-2026/code title="ODP to JPEG Java Conversion Source Code" >}}
```java
Presentation presentation = new Presentation("template.odp");
try {
    for (ISlide slide : presentation.getSlides()) {
        IImage slideImage = slide.getImage(1f, 1f);
        try {
            slideImage.save("slide_" + slide.getSlideNumber() + ".jpeg", ImageFormat.Jpeg);
        } finally {
            slideImage.dispose();
        }
    }
} finally {
    presentation.dispose();
}
```
