{{< blocks/products/pf/slides-design-update-10-2026/code title="Merge PPTX Files - Java" >}}
```java
Presentation destinationPresentation = new Presentation("presentation1.pptx");
try {
    Presentation sourcePresentation = new Presentation("presentation2.pptx");
    try {
        for (ISlide slide : sourcePresentation.getSlides()) {
            destinationPresentation.getSlides().addClone(slide);
        }
    } finally {
        sourcePresentation.dispose();
    }

    destinationPresentation.save("merged-presentation.pptx", SaveFormat.Pptx);
} finally {
    destinationPresentation.dispose();
}
```
