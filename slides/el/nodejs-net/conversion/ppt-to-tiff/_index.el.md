{{< blocks/products/pf/slides-design-update-10-2026/code title="Κώδικας JavaScript για μετατροπή PPT σε TIFF" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.tiff", SaveFormat.Tiff);
}
finally
{
    if (pres != null) pres.dispose();
}
```
