import asposeslides

jpype.startJVM()

from asposeslides.api import Presentation, SaveFormat
from javax.imageio import ImageIO
from java.io import File

pres = Presentation("PowerPoint.potx");

pres.save("output.fodp", SaveFormat.Fodp);

jpype.shutdownJVM()
