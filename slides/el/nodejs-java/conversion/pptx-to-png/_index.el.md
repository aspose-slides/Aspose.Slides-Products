Για να μετατρέψετε το PPTX σε PNG, θα χρειαστεί να δημιουργήσετε την Παρουσίαση από το αρχείο PPTX και να την αποθηκεύσετε ως PNG.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Κώδικας Node.js για μετατροπή PPTX σε PNG" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptx");
try
{
    for(var i = 0; i < pres.getSlides().size(); i++)
    {
        var sld = pres.getSlides().get_Item(i);
        var bi = sld.getThumbnail(2, 2);
        var outputfile = java.newInstanceSync("java.io.File", "slide_" + sld.getSlideNumber() + ".png");
        java.callStaticMethod("javax.imageio.ImageIO", "write", bi, "png", outputfile);
    }
}
finally
{
    if (pres != null) pres.dispose();
}
```
