Για να μετατρέψετε το PDF σε PPSX, θα χρειαστεί να δημιουργήσετε την Παρουσίαση από το αρχείο PDF και να την αποθηκεύσετε ως PPSX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Κώδικας Node.js για μετατροπή PDF σε PPSX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.ppsx", aspose.slides.SaveFormat.Ppsx);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
