Để chuyển đổi POT sang PPTX, bạn cần tạo Bản trình bày từ tệp POT và lưu dưới dạng PPTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Mã Node.js để chuyển đổi POT thành PPTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pot");
try
{
    pres.save("output.pptx", aspose.slides.SaveFormat.Pptx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
