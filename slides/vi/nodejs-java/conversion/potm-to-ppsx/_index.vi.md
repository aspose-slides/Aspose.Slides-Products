Để chuyển đổi POTM sang PPSX, bạn cần tạo Bản trình bày từ tệp POTM và lưu dưới dạng PPSX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Mã Node.js để chuyển đổi POTM thành PPSX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.potm");
try
{
    pres.save("output.ppsx", aspose.slides.SaveFormat.Ppsx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
