Để chuyển đổi PPS sang PPT, bạn cần tạo Bản trình bày từ tệp PPS và lưu dưới dạng PPT.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Mã Node.js để chuyển đổi PPS thành PPT" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pps");
try
{
    pres.save("output.ppt", aspose.slides.SaveFormat.Ppt);
}
finally
{
    if (pres != null) pres.dispose();
}
```
