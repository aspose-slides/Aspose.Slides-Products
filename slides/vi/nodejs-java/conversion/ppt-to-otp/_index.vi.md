Để chuyển đổi PPT sang OTP, bạn cần tạo Bản trình bày từ tệp PPT và lưu dưới dạng OTP.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Mã Node.js để chuyển đổi PPT thành OTP" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.otp", aspose.slides.SaveFormat.Otp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
