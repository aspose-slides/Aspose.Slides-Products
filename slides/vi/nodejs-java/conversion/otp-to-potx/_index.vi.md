Để chuyển đổi OTP sang POTX, bạn cần tạo Bản trình bày từ tệp OTP và lưu dưới dạng POTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Mã Node.js để chuyển đổi OTP thành POTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.otp");
try
{
    pres.save("output.potx", aspose.slides.SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
