PPT を HTML に変換するには、PPT ファイルからプレゼンテーションを作成し、それを HTML として保存する必要があります。

{{< blocks/products/pf/slides-design-update-10-2026/code title="PPT を HTML に変換するための Node.js コード" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.html", aspose.slides.SaveFormat.Html5);
}
finally
{
    if (pres != null) pres.dispose();
}
```
