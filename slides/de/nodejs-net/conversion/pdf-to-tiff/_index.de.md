Um das PDF in TIFF zu konvertieren, müssen Sie eine Präsentation aus der PDF-Datei erstellen und diese als TIFF speichern.

{{< blocks/products/pf/slides-design-update-10-2026/code title="JavaScript-Code zum Konvertieren von PDF in TIFF" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat, ShapeType } = asposeSlides;

var pres = new Presentation();
try
{
    pres.slides.removeAt(0);
    var slides = pres.slides.addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.tiff", SaveFormat.Tiff);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
