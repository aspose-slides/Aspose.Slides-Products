Um das PPSM in PPTX zu konvertieren, müssen Sie eine Präsentation aus der PPSM-Datei erstellen und diese als PPTX speichern.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-Code zum Konvertieren von PPSM in PPTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppsm");
try
{
    pres.save("output.pptx", aspose.slides.SaveFormat.Pptx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
