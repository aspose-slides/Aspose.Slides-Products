Um das PPT in POTM zu konvertieren, müssen Sie eine Präsentation aus der PPT-Datei erstellen und diese als POTM speichern.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-Code zum Konvertieren von PPT in POTM" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.potm", aspose.slides.SaveFormat.Potm);
}
finally
{
    if (pres != null) pres.dispose();
}
```
