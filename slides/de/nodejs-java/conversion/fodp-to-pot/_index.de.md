Um das FODP in POT zu konvertieren, müssen Sie eine Präsentation aus der FODP-Datei erstellen und diese als POT speichern.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-Code zum Konvertieren von FODP in POT" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.fodp");
try
{
    pres.save("output.pot", aspose.slides.SaveFormat.Pot);
}
finally
{
    if (pres != null) pres.dispose();
}
```
