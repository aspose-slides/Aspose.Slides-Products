Um das OTP in GIF zu konvertieren, müssen Sie eine Präsentation aus der OTP-Datei erstellen und diese als GIF speichern.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js-Code zum Konvertieren von OTP in GIF" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.otp");
try
{
    pres.save("output.gif", aspose.slides.SaveFormat.Gif);
}
finally
{
    if (pres != null) pres.dispose();
}
```
