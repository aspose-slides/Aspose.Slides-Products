PPSX dosyasını TIFF biçimine dönüştürmek için, PPSX dosyasından Sunum oluşturmanız ve bunu TIFF olarak kaydetmeniz gerekir.

{{< blocks/products/pf/slides-design-update-10-2026/code title="PPSX biçimini TIFF biçimine dönüştürmek için Node.js kodu" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppsx");
try
{
    pres.save("output.tiff", aspose.slides.SaveFormat.Tiff);
}
finally
{
    if (pres != null) pres.dispose();
}
```
