PPTM dosyasını POT biçimine dönüştürmek için, PPTM dosyasından Sunum oluşturmanız ve bunu POT olarak kaydetmeniz gerekir.

{{< blocks/products/pf/slides-design-update-10-2026/code title="PPTM biçimini POT biçimine dönüştürmek için Node.js kodu" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.pptm");
try
{
    pres.save("output.pot", aspose.slides.SaveFormat.Pot);
}
finally
{
    if (pres != null) pres.dispose();
}
```
