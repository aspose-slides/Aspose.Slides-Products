{{< blocks/products/pf/slides-design-update-10-2026/code title="PPT biçimini POT biçimine dönüştürmek için JavaScript kodu" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.ppt");
try
{
    pres.save("output.pot", SaveFormat.Pot);
}
finally
{
    if (pres != null) pres.dispose();
}
```
