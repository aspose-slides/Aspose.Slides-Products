{{< blocks/products/pf/slides-design-update-10-2026/code title="Python oktatóanyag a PPTM PPSM formátumba való konvertálásához" >}}
```python

import jpype
import asposeslides

jpype.startJVM()

from asposeslides.api import Presentation, SaveFormat
from javax.imageio import ImageIO
from java.io import File

pres = Presentation("PowerPoint.pptm");

pres.save("output.ppsm", SaveFormat.Ppsm);

jpype.shutdownJVM()
```
