{{< blocks/products/pf/slides-design-update-10-2026/code title="JavaScript kód a FODP TIFF formátumba való konvertálásához" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.fodp");
try
{
    pres.save("output.tiff", SaveFormat.Tiff);
}
finally
{
    if (pres != null) pres.dispose();
}
```
