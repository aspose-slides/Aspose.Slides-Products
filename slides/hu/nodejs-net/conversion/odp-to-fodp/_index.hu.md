{{< blocks/products/pf/slides-design-update-10-2026/code title="JavaScript kód a ODP FODP formátumba való konvertálásához" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.odp");
try
{
    pres.save("output.fodp", SaveFormat.Fodp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
