{{< blocks/products/pf/slides-design-update-10-2026/code title="JavaScript kód a PPTM OTP formátumba való konvertálásához" >}}
```javascript

const fs = require('fs');
const asposeSlides = require('aspose.slides.via.net');
const { Presentation, SaveFormat } = asposeSlides;
var pres = new Presentation("welcome-to-powerpoint.pptm");
try
{
    pres.save("output.otp", SaveFormat.Otp);
}
finally
{
    if (pres != null) pres.dispose();
}
```
