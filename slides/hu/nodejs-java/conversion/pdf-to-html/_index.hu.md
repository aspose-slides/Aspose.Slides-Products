A PDF formátum HTML formátumra konvertálásához létre kell hoznia egy prezentációt a PDF fájlból, és el kell mentenie HTML néven.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Node.js kód a PDF HTML formátumba való konvertálásához" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.html", aspose.slides.SaveFormat.Html5);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
