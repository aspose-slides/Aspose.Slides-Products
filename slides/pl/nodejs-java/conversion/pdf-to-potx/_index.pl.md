Aby przekonwertować plik PDF na POTX, musisz utworzyć prezentację z pliku PDF i zapisać ją jako POTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Kod Node.js do konwersji PDF na POTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation();
try
{
    pres.getSlides().removeAt(0);
    pres.getSlides().addFromPdf("welcome-to-powerpoint.pdf");

    pres.save("output.potx", aspose.slides.SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
} 

```
