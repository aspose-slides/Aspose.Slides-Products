Aby przekonwertować plik PPSX na HTML, musisz utworzyć prezentację z pliku PPSX i zapisać ją jako HTML.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Kod Node.js do konwersji PPSX na HTML" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppsx");
try
{
    pres.save("output.html", aspose.slides.SaveFormat.Html5);
}
finally
{
    if (pres != null) pres.dispose();
}
```
