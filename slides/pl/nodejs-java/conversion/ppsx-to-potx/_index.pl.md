Aby przekonwertować plik PPSX na POTX, musisz utworzyć prezentację z pliku PPSX i zapisać ją jako POTX.

{{< blocks/products/pf/slides-design-update-10-2026/code title="Kod Node.js do konwersji PPSX na POTX" >}}
```javascript

var aspose = aspose || {};

aspose.slides = require("aspose.slides.via.java");

var pres = new aspose.slides.Presentation("welcome-to-powerpoint.ppsx");
try
{
    pres.save("output.potx", aspose.slides.SaveFormat.Potx);
}
finally
{
    if (pres != null) pres.dispose();
}
```
